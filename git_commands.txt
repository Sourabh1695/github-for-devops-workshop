# git initialization
  git init

# Check whether files are tracked or untracked
  git status

# Add file to git (staging)
  git add <file_name>

# File commit (converting untracked files to tracked files)
  git commit -m "added testing files"

# Restore file from git (restoring files from git)
  git restore <file_name>

# adding repo from github into local (1 time command)
  git remote add origin https://github.com/Sourabh1695/github-for-devops-workshop
 
# Pushing into github 
  git push origin main
